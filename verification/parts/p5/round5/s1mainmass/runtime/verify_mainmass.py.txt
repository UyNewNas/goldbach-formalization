def now() -> str:
    return dt.datetime.now(dt.timezone.utc).isoformat()


def digest(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        for block in iter(lambda: f.read(1024 * 1024), b""):
            h.update(block)
    return h.hexdigest()


def file_record(path: Path) -> dict:
    return {"path": str(path), "sha256": digest(path), "bytes": path.stat().st_size}


def git(repo: Path, *args: str) -> str:
    return subprocess.check_output(["git", "-C", str(repo), *args], text=True).strip()


def source(module: str) -> Path:
    return GOLDBACH / (module.replace(".", "/") + ".lean")


def output(module: str) -> Path:
    return OUTROOT / (module.replace(".", "/") + ".olean")


def imports(path: Path) -> list[str]:
    result = []
    for line in path.read_text().splitlines():
        if line.startswith("import "):
            result.extend(line.removeprefix("import ").split())
    return result


def environment() -> dict[str, str]:
    roots = [OUTROOT, SHARED / ".lake/build/lib/lean"]
    src_roots = [REPO, SHARED]
    for name in PACKAGES:
        package = SHARED / ".lake/packages" / name
        roots.append(package / ".lake/build/lib/lean")
        src_roots.append(package)
    env = os.environ.copy()
    env.update({
        "LEAN_SYSROOT": str(TOOLROOT),
        "LAKE_HOME": str(TOOLROOT),
        "LEAN_NUM_THREADS": "1",
        "LEAN_PATH": ":".join(str(x) for x in roots),
        "LEAN_SRC_PATH": ":".join(str(x) for x in src_roots),
        "PATH": str(TOOLROOT / "bin") + ":" + env.get("PATH", ""),
    })
    return env


def save(path: Path, value: object) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, indent=2, ensure_ascii=False) + "\n")


def preflight() -> dict:
    assert git(REPO, "rev-parse", "HEAD") == START_HEAD
    target_source = source(TARGET)
    direct_source = source(DIRECT)
    assert digest(target_source) == TARGET_SHA
    assert digest(direct_source) == DIRECT_SHA
    assert imports(target_source) == [DIRECT]
    packages = []
    for name, rev in PACKAGES.items():
        root = SHARED / ".lake/packages" / name
        actual = git(root, "rev-parse", "HEAD")
        changes = git(root, "diff", "--name-only", "HEAD").splitlines()
        assert actual == rev and changes == [], (name, actual, changes)
        packages.append({"package": name, "revision": actual, "tracked_changes": changes})
    binaries = {name: file_record(TOOLROOT / "bin" / name)
                for name in ["lean", "lake", "leanchecker", "leantar"]}
    driver = file_record(DRIVER / "lakefile.lean")
    manifest = file_record(DRIVER / "lake-manifest.json")
    result = {
        "observed_utc": now(), "starting_head": START_HEAD, "common_baseline": BASELINE,
        "target": TARGET, "target_source": file_record(target_source),
        "direct_imports": [{"module": DIRECT, "source": file_record(direct_source),
                            "objects_before": [file_record(output(DIRECT))] if output(DIRECT).exists() else []}],
        "target_objects_before": [file_record(output(TARGET))] if output(TARGET).exists() else [],
        "imports_exact": imports(target_source), "packages": packages,
        "toolchain": {"name": "leanprover/lean4:v4.33.0-rc1", "githash": LEAN_GITHASH,
                      "binaries": binaries},
        "frontend_driver": driver, "frontend_manifest": manifest,
        "audit_source": file_record(REPO / "verification/parts/p5/S1MainMassAudit.lean"),
        "LEAN_PATH": environment()["LEAN_PATH"],
        "scope": "Compile one necessary missing compatibility import, then accept S1MainMass only; no closure replay/audit.",
    }
    save(RUNTIME / "preflight.json", result)
    return result


def compile_command(module: str, src: Path, out: Path) -> list[str]:
    return [str(TOOLROOT / "bin/lake"), "-d", str(DRIVER), "-f", "lakefile.lean",
            "script", "run", "compileOne", str(src), module, str(out), "strict"]


def restore_certified_import() -> dict:
    """Restore the hash-certified local namespace closure into P5 output."""
    prior_path = REPO / "verification/parts/p5/round3/s1carrier/preflight.json"
    prior = json.loads(prior_path.read_text())
    restored_rows = []
    direct_certificate = None
    for certificate in prior["local_compiled_imports"]:
        module = certificate["module"]
        source_record = certificate["source"]
        result = certificate["certificate"]["result"]
        assert result["returncode"] == 0 and result["trustLevel"] == 0
        source_path = Path(source_record["path"])
        assert source_path.exists() and digest(source_path) == source_record["sha256"]
        cached_base = Path(source_record["package_root"]) / ".lake/build/lib/lean" / (module.replace(".", "/") + ".olean")
        expected = certificate["objects"][0]
        if module == DIRECT and not cached_base.exists():
            direct_certificate = {
                "module": module, "source_sha256": source_record["sha256"],
                "historical_object_sha256": expected["sha256"],
                "historical_object_bytes": expected["bytes"],
                "historical_returncode": result["returncode"],
                "historical_trustLevel": result["trustLevel"],
                "historical_raw_log_sha256": certificate["certificate"]["raw_log_sha256"],
                "current_cached_object_missing": True,
                "action": "compile_strict_in_this_batch_after_restoring_certified_imports",
            }
            continue
        object_rows = []
        for expected_file in certificate["objects"]:
            cached = cached_base.parent / Path(expected_file["path"]).name
            assert cached.exists() and digest(cached) == expected_file["sha256"]
            assert cached.stat().st_size == expected_file["bytes"]
            restored = output(module).parent / cached.name
            restored.parent.mkdir(parents=True, exist_ok=True)
            shutil.copyfile(cached, restored)
            assert digest(restored) == expected_file["sha256"]
            object_rows.append({"path": str(restored), "sha256": expected_file["sha256"],
                                "bytes": expected_file["bytes"]})
        restored_rows.append({
            "module": module, "source_sha256": source_record["sha256"],
            "object_sha256": expected["sha256"], "object_bytes": expected["bytes"],
            "restored_objects": object_rows, "historical_returncode": result["returncode"],
            "historical_trustLevel": result["trustLevel"],
            "historical_raw_log_sha256": certificate["certificate"]["raw_log_sha256"],
        })
    matches = [x for x in restored_rows if x["module"] == RESTORED_IMPORT]
    assert len(matches) == 1
    assert matches[0]["source_sha256"] == RESTORED_SOURCE_SHA
    assert matches[0]["object_sha256"] == RESTORED_OBJECT_SHA
    assert direct_certificate is not None and direct_certificate["source_sha256"] == DIRECT_SHA
    manifest_path = RUNTIME / "restoration-manifest.json"
    save(manifest_path, restored_rows)
    record = {
        "action": "restore_hash_certified_local_namespace_closure",
        "restored_module_count": len(restored_rows),
        "manifest": {"path": str(manifest_path), "sha256": digest(manifest_path)},
        "direct_import": direct_certificate,
        "named_distribution_witness": matches[0],
        "historical_certificate": {
            "path": str(prior_path), "sha256": digest(prior_path),
        },
        "scope": "Byte-identical restoration of 95 prior P5 hash-certified local objects; none was recompiled, replayed, or re-audited in this batch. The missing compatibility import is compiled strictly in this batch. Official-cache modules stay in their unchanged package roots."
    }
    save(RUNTIME / "restoration.json", record)
    return record


def run(action: str, command: list[str]) -> dict:
    stamp = dt.datetime.now(dt.timezone.utc).strftime("%Y%m%dT%H%M%S%fZ")
    log = LOGS / f"{stamp}-{action}.log"
    LOGS.mkdir(parents=True, exist_ok=True)
    start = time.monotonic()
    with log.open("wb") as f:
        proc = subprocess.run(command, cwd=REPO, env=environment(), stdout=f, stderr=subprocess.STDOUT)
    elapsed = time.monotonic() - start
    record = {"action": action, "command": command, "cwd": str(REPO),
              "started_utc": stamp, "returncode": proc.returncode,
              "elapsed_seconds": round(elapsed, 3), "raw_log": str(log),
              "raw_log_sha256": digest(log), "trustLevel": 0 if action != "replay" else None,
              "warningAsError": action != "replay"}
    print(json.dumps(record, ensure_ascii=False), flush=True)
    if proc.returncode != 0:
        print(log.read_text(errors="replace")[-12000:], file=sys.stderr, flush=True)
        raise SystemExit(proc.returncode)
    return record


def main() -> None:
    pf = preflight()
    restoration = restore_certified_import()
    records = []
    dep_out = output(DIRECT)
    if not dep_out.exists():
        records.append(run("dependency-compile", compile_command(DIRECT, source(DIRECT), dep_out)))
    target_out = output(TARGET)
    records.append(run("compile", compile_command(TARGET, source(TARGET), target_out)))
    records.append(run("replay", [str(TOOLROOT / "bin/leanchecker"), "--verbose", TARGET]))
    audit_src = REPO / "verification/parts/p5/S1MainMassAudit.lean"
    audit_out = output(AUDIT)
    records.append(run("audit", compile_command(AUDIT, audit_src, audit_out)))
    assert digest(source(TARGET)) == TARGET_SHA and digest(source(DIRECT)) == DIRECT_SHA
    result = {
        "status": "GATES_PASSED", "completed_utc": now(), "preflight_sha256": digest(RUNTIME / "preflight.json"),
        "restoration": restoration, "commands": records, "target_object": file_record(target_out),
        "direct_import_object": file_record(dep_out), "audit_object": file_record(audit_out),
        "target_source_after": file_record(source(TARGET)), "direct_import_source_after": file_record(source(DIRECT)),
        "audit_source_after": file_record(audit_src), "starting_head_after": git(REPO, "rev-parse", "HEAD"),
        "packages_after": pf["packages"],
    }
    save(RUNTIME / "result.json", result)
    print(json.dumps({"status": result["status"], "target_object": result["target_object"],
                      "direct_import_object": result["direct_import_object"],
                      "audit_object": result["audit_object"]}, ensure_ascii=False), flush=True)


if __name__ == "__main__":
    main()
