"""Assemble a receipt only after all three actual target gates have passed."""
from pathlib import Path
from collections import Counter
import base64, gzip, hashlib, json, shlex, shutil
import verify_s1lowerrosser as m

s, d, E = m.s, m.d, m.EVIDENCE
LOGS = m.HERE / 'logs'
RESTORE = m.HERE / 'restore'
records = [json.loads(line) for line in (LOGS/'results.jsonl').read_text().splitlines()]
gates = {}
for action in ['compile', 'replay', 'audit']:
    matches = [r for r in records if r['action'] == action]
    assert matches, ('missing gate', action)
    r = matches[-1]
    assert r['returncode'] == 0 and r['source_unchanged'] and r['imports_unchanged'] and r['guards_unchanged']
    if action != 'replay':
        assert r['trustLevel'] == 0 and r['warningAsError'] and not r['metadata_adapter']
        assert 'FRONTEND_OK module=' + r['module'] + ' trustLevel=0' in Path(r['log']).read_text()
    assert d.digest(Path(r['log'])) == r['raw_log_sha256']
    gates[action] = r
assert gates['compile']['source_sha256_after'] == s.SOURCE_SHA
assert gates['compile']['objects_after'] == gates['replay']['objects_before'] == gates['replay']['objects_after']
assert gates['audit']['imports_before'][0]['objects'] == gates['compile']['objects_after']
pre = json.loads((LOGS/'preflight.json').read_text())
assert not pre['missing_import_objects']

audit_lines = Path(gates['audit']['log']).read_text().splitlines()
marker = 'CHEN_P5_S1LOWERROSSER_AUDIT_DECL '
decls = [json.loads(line[len(marker):]) for line in audit_lines if line.startswith(marker)]
count_lines = [line for line in audit_lines if line.startswith('CHEN_P5_S1LOWERROSSER_AUDIT_COUNT ')]
assert len(count_lines) == 1 and int(count_lines[0].split()[-1]) == len(decls) and decls
assert len({x['name'] for x in decls}) == len(decls)
allowed = {'propext', 'Classical.choice', 'Quot.sound'}
for x in decls:
    assert x['module'] == s.TARGET and not x['is_unsafe'] and not x['is_axiom']
    assert set(x['axioms']) <= allowed and x['type_explicit'] and x['type_expr_repr']
public = [x for x in decls if x['requested_public_theorem']]
assert len(public) == 2
assert {x['name'].split('.')[-1] for x in public} == {
    'goldbachS1BoundingSieve_lowerErrSum_le_prefixSum', 'goldbachS1_lowerRosser_mainTerm_sub_prefix_le'}
assert all(x['kind'] == 'theorem' for x in public)
types = {x['name'].split('.')[-1]:x for x in public}
assert '@Int.cast' in types['goldbachS1_lowerRosser_mainTerm_sub_prefix_le']['type_explicit']

restore_rows = [json.loads(line) for line in (RESTORE/'results.jsonl').read_text().splitlines()]
restore_plan = json.loads((RESTORE/'plan.json').read_text())
restore_done = json.loads((RESTORE/'serial-completion.json').read_text())
assert restore_done['status'] == 'EXACT_MISSING_IMPORT_OBJECTS_RESTORED'
assert {x['module'] for x in restore_rows} == {x['module'] for x in restore_plan['source_order']}
for row in restore_rows:
    assert row['historical_object_sha256_reproduced'] and row['source_unchanged'] and row['imports_unchanged'] and row['guards_unchanged']
    if row['action'] == 'restore_required_import':
        assert row['returncode'] == 0 and row['trustLevel'] == 0 and not row['metadata_adapter']
    else:
        assert row['action'] == 'reuse_matching_restored_import' and not row['frontend_run_by_this_copy']
restore_compiled = [x for x in restore_rows if x['action'] == 'restore_required_import']
restore_copied = [x for x in restore_rows if x['action'] == 'reuse_matching_restored_import']

E.mkdir(parents=True, exist_ok=True)
for name in ['raw', 'runtime', 'restoration', 'restoration/raw', 'source', 'objects']:
    (E/name).mkdir(parents=True, exist_ok=True)
shutil.copy2(LOGS/'preflight.json', E/'preflight.json')
s.save(E/'commands-and-results.json', records)
for r in records:
    shutil.copy2(r['log'], E/'raw'/Path(r['log']).name)
for p in RESTORE.iterdir():
    if p.is_file() and p.name != 'active.json':
        dest = E/'restoration/raw'/p.name if p.suffix == '.log' else E/'restoration'/p.name
        shutil.copy2(p, dest)
for name in ['recovery-required.json', 'setup-diagnostics.json']:
    if (LOGS/name).exists():
        shutil.copy2(LOGS/name, E/'restoration'/name)
for p in m.HERE.glob('*.py'):

shutil.copy2(d.DRIVER/'lakefile.lean', E/'runtime/frontend-driver.lakefile.lean.txt')

    x['name']+'\nkind: '+x['kind']+'\naxioms: '+', '.join(x['axioms'])+'\n\nTYPE:\n'+x['type']+'\n\nEXPLICIT TYPE:\n'+x['type_explicit'] for x in decls)+'\n')
(E/'public-theorem-types.lean.txt').write_text(header+'\n\n'.join(
    x['name']+' :\n'+x['type']+'\n\n-- Complete explicit compiler type:\n'+x['type_explicit'] for x in public)+'\n')
cmds = ['#!/usr/bin/env bash', 'set -euo pipefail', '# Actual target commands, in execution order. See JSON for cwd, environment and results.',
    '# The fixed frontend driver calls Lean.Elab.runFrontend at trustLevel=0.',
    'export LEAN_SYSROOT='+shlex.quote(str(d.TOOLROOT)), 'export LAKE_HOME='+shlex.quote(str(d.TOOLROOT)),
    'export LEAN_NUM_THREADS=1', 'export PATH='+shlex.quote(str(d.TOOLROOT/'bin'))+':"$PATH"',
    'export LEAN_PATH='+shlex.quote(d.environment()['LEAN_PATH']),
    'export LEAN_SRC_PATH='+shlex.quote(d.environment()['LEAN_SRC_PATH']), 'cd '+shlex.quote(str(d.REPO))]
for r in records:
    cmds += ['', '# '+r['action']+': exit '+str(r['returncode'])+', '+str(r['elapsed_seconds'])+' seconds', shlex.join(r['command'])]
(E/'commands.sh').write_text('\n'.join(cmds)+'\n')
union = sorted({a for x in decls for a in x['axioms']})
object_bytes=Path(gates['compile']['objects_after'][0]['path']).read_bytes()
object_packed=gzip.compress(object_bytes,mtime=0)
object_encoded=base64.encodebytes(object_packed)
assert gzip.decompress(base64.decodebytes(object_encoded))==object_bytes
object_archive=E/'objects/LiLiuGoldbachS1LowerRosser.olean.gz.b64'
object_archive.write_bytes(object_encoded)
object_archive_record={'path':'objects/'+object_archive.name,'format':'base64 of gzip, mtime=0',
    'transport_sha256':d.digest(object_archive),'gzip_sha256':hashlib.sha256(object_packed).hexdigest(),
    'decoded_object_sha256':hashlib.sha256(object_bytes).hexdigest(),
    'decoded_object_bytes':len(object_bytes),'roundtrip_bytes_verified':True,
    'scope':'Transport copy of the already strictly compiled, replayed and audited target object; reuse still requires matching fixed source/toolchain/import identities.'}

    'audit':{'method':'Defining-module environment inventory and bundled Lean.collectAxioms',

        'source_sha256':gates['audit']['source_sha256_after'], 'compiled_objects':gates['audit']['objects_after'],
        'declaration_count':len(decls),'requested_public_theorem_count':len(public),
        'kind_counts':dict(Counter(x['kind'] for x in decls)),

        'private_and_generated_included':True,'unsafe_declarations':[], 'own_axiom_declarations':[], 'forbidden_axioms':[],
        'transitive_axiom_union':union,'types_status':'COMPILER_CONFIRMED_FROM_ELABORATED_ENVIRONMENT',

        'S1Carrier_replayed_this_batch':False,'S1Carrier_reaudited_this_batch':False,
        'previous_S1Carrier_gates_reused':pre['reused_S1Carrier']['gates'],

for row in context['supplier_graph']:
    if row['module']==s.TARGET:row['status']='STRICT_REPLAY_FULL_AXIOM_AUDIT_PASSED'
