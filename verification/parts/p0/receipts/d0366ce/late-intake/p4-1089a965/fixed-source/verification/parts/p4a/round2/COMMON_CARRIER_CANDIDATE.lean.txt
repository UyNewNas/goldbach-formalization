theorem smallRoughRho_eq_of_rough_subset
    (small Rraw R : Finset ℕ)
    (raw betaMinus betaPlus : ℕ → ℝ)
    (hsub : Rraw ⊆ R)
    (hzero : ∀ d ∈ R, d ∉ Rraw → raw d = 0) :
    smallRoughRho small R raw betaMinus betaPlus =
      smallRoughRho small Rraw raw betaMinus betaPlus := by
  classical
  funext q
  simp only [smallRoughRho, Finset.sum_filter, smallRoughPairs, Finset.sum_product]
  apply Finset.sum_congr rfl
  intro a _ha
  symm
  apply Finset.sum_subset hsub
  intro d hd hdRaw
  simp [smallRoughPairCoefficient, signAttachedCoefficient, hzero d hd hdRaw]

theorem sum_smallRoughRho_mul_eq_of_rough_subset
    (small Rraw R : Finset ℕ)
    (raw betaMinus betaPlus F : ℕ → ℝ)
    (hsub : Rraw ⊆ R)
    (hzero : ∀ d ∈ R, d ∉ Rraw → raw d = 0) :
    (∑ q ∈ smallRoughModuli small R,
      smallRoughRho small R raw betaMinus betaPlus q * F q) =
    ∑ q ∈ smallRoughModuli small Rraw,
      smallRoughRho small Rraw raw betaMinus betaPlus q * F q := by
  classical
  have hmod :
      smallRoughModuli small Rraw ⊆ smallRoughModuli small R := by
    intro q hq
    rcases Finset.mem_image.mp hq with ⟨x, hx, rfl⟩
    refine Finset.mem_image.mpr ⟨x, ?_, rfl⟩
    exact Finset.mem_product.mpr
      ⟨(Finset.mem_product.mp hx).1, hsub ((Finset.mem_product.mp hx).2)⟩
  rw [smallRoughRho_eq_of_rough_subset
    small Rraw R raw betaMinus betaPlus hsub hzero]
  symm
  apply Finset.sum_subset hmod
  intro q _hq hqRaw
  simp only [smallRoughRho_eq_zero_of_not_mem
    small Rraw raw betaMinus betaPlus hqRaw, zero_mul]

theorem actualSmallRoughRho_eq_rough_union
    (retainOdd : List ℕ → Bool) (N cap : ℕ) (w z L : ℝ)
    (extra : Finset ℕ) :
    smallRoughRho (actualSmallRosserSupport N cap w)
      (fixedRoughSupport retainOdd N cap w z ∪ extra)
      (fixedRoughCoefficient retainOdd N cap w z)
      (actualSmallRosserMinus N cap w L) (actualSmallRosserPlus N cap w L) =
    actualSmallRoughRho retainOdd N cap w z L := by
  unfold actualSmallRoughRho
  apply smallRoughRho_eq_of_rough_subset
  · exact Finset.subset_union_left
  · intro d _hd hnot
    change (oddFirstExitRetainedCoefficient retainOdd (fun _ => 1)
      (roughPrimeUniverse N cap w z) d : ℝ) = 0
    exact_mod_cast
      (oddFirstExitRetainedCoefficient_eq_zero_of_not_mem
        retainOdd (fun _ => 1) (roughPrimeUniverse N cap w z) d hnot)

theorem supportedDivisorSum_eq_of_subset
    (Rraw R : Finset ℕ) (raw : ℕ → ℝ) (n : ℕ)
    (hsub : Rraw ⊆ R)
    (hzero : ∀ d ∈ R, d ∉ Rraw → raw d = 0) :
    supportedDivisorSum R raw n = supportedDivisorSum Rraw raw n := by
  classical
  simp only [supportedDivisorSum, Finset.sum_filter]
  symm
  apply Finset.sum_subset hsub
  intro d hd hdRaw
  simp [hzero d hd hdRaw]

/-- Uncompiled usage example: the actual raw physical density is unchanged
when its rough carrier is enlarged by arbitrary extra indices. -/
example (retainOdd : List ℕ → Bool) (N cap : ℕ) (w z L : ℝ)
    (extra : Finset ℕ) :
    physicalDensitySum
      (smallRoughModuli (actualSmallRosserSupport N cap w)
        (fixedRoughSupport retainOdd N cap w z ∪ extra))
      (smallRoughRho (actualSmallRosserSupport N cap w)
        (fixedRoughSupport retainOdd N cap w z ∪ extra)
        (fixedRoughCoefficient retainOdd N cap w z)
        (actualSmallRosserMinus N cap w L) (actualSmallRosserPlus N cap w L)) =
    physicalDensitySum (actualSmallRoughModuli retainOdd N cap w z)
      (actualSmallRoughRho retainOdd N cap w z L) := by
  have hzero : ∀ d ∈ fixedRoughSupport retainOdd N cap w z ∪ extra,
      d ∉ fixedRoughSupport retainOdd N cap w z →
        fixedRoughCoefficient retainOdd N cap w z d = 0 := by
    intro d _hd hnot
    change (oddFirstExitRetainedCoefficient retainOdd (fun _ => 1)
      (roughPrimeUniverse N cap w z) d : ℝ) = 0
    exact_mod_cast
      (oddFirstExitRetainedCoefficient_eq_zero_of_not_mem
        retainOdd (fun _ => 1) (roughPrimeUniverse N cap w z) d hnot)
  unfold actualSmallRoughModuli actualSmallRoughRho
  simpa only [physicalDensitySum, div_eq_mul_inv] using
    (sum_smallRoughRho_mul_eq_of_rough_subset
      (actualSmallRosserSupport N cap w)
      (fixedRoughSupport retainOdd N cap w z)
      (fixedRoughSupport retainOdd N cap w z ∪ extra)
      (fixedRoughCoefficient retainOdd N cap w z)
      (actualSmallRosserMinus N cap w L) (actualSmallRosserPlus N cap w L)
      (fun q => (Nat.totient q : ℝ)⁻¹)
      Finset.subset_union_left hzero)
