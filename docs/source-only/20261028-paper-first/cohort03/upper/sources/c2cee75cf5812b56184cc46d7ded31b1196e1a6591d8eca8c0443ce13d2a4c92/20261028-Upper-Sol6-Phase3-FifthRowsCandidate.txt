abbrev archive := Data.P1HistoricalFifthSavedTableData.rows
def savedRecords : Finset FifthWireRow := archive.toList.toFinset
def savedRows5 (q : ℕ) : Finset (FifthRow (refinedMesh q)) :=
  decodedFamily savedRecords q

-- The image boundary is explicit: masks remain in savedRows5, while J5
-- records owners only. Multiple masks on one owner are never first-matched.
def savedJ5 (q : ℕ) : Finset (RefinedCell q) :=
  (savedRows5 q).image FifthRow.cell

def savedOwners (e : ℚ) (chi omega : ℝ) (q : ℕ) : Finset (RefinedCell q) :=
  P1OriginalCoarseFifthContinuation.selectedFifthCells
    e chi (refinedMesh q) omega (savedJ5 q)

def savedSelectedRows (e : ℚ) (chi omega : ℝ) (q : ℕ) :
    Finset (FifthRow (refinedMesh q)) :=
  (savedRows5 q).filter (fun r => r.cell ∈ savedOwners e chi omega q)

theorem mem_savedJ5 (q : ℕ) (j : RefinedCell q) :
    j ∈ savedJ5 q ↔ ∃ raw ∈ archive.toList,
      ∃ r ∈ decodeFifthRows raw (refinedMesh q), r.cell = j := by
  constructor
  · intro hj
    obtain ⟨r, hr, hcell⟩ := Finset.mem_image.mp hj
    obtain ⟨raw, hraw, hdecode⟩ := (mem_decodedFamily savedRecords q r).mp hr
    exact ⟨raw, List.mem_toFinset.mp hraw, r, hdecode, hcell⟩
  · rintro ⟨raw, hraw, r, hr, hcell⟩
    exact Finset.mem_image.mpr ⟨r,
      (mem_decodedFamily savedRecords q r).mpr
        ⟨raw, List.mem_toFinset.mpr hraw, hr⟩, hcell⟩

-- This is the exact finite original owner filter; the lower endpoint
-- rough test is inclusive. No firstStop cap or initial<=third claim occurs.
theorem mem_savedOwners (e : ℚ) (chi omega : ℝ) (q : ℕ) (j : RefinedCell q) :
    j ∈ savedOwners e chi omega q ↔
      (∃ raw ∈ archive.toList, ∃ r ∈ decodeFifthRows raw (refinedMesh q), r.cell = j) ∧
      historicalPrefixProject (by omega : 3 ≤ 5) j ∈
        HistoricalThirdCoarse.cells (e : ℝ) chi (refinedMesh q) ∧
      omega ≤ (qLogPrimeMeshLo 0 (19 / 96) (refinedMesh q)
        (historicalPrefixProject (by omega : 3 ≤ 5) j) 2 : ℝ) := by
  rw [savedOwners, P1OriginalCoarseFifthContinuation.mem_selectedFifthCells,
    mem_savedJ5]

theorem mem_savedSelectedRows (e : ℚ) (chi omega : ℝ) (q : ℕ)
    (r : FifthRow (refinedMesh q)) :
    r ∈ savedSelectedRows e chi omega q ↔

      r.cell ∈ savedOwners e chi omega q := by

-- Removing unselected data rows changes no cap on ANY selected owner.
-- Both maxima keep the identical originalVariableFourthProfile seed.
theorem actualNext5_saved_selected_rows (e : ℚ) (chi omega : ℝ) (q : ℕ)
    (rows4 : Finset ProducerRow) (licensed4 : ProducerRow → Prop)
    (j : RefinedCell q) (hj : j ∈ savedOwners e chi omega q) (t : Fin 5 → ℝ) :
    actualNext5 e chi (refinedMesh q) omega j rows4 licensed4 (savedRows5 q) t =
      actualNext5 e chi (refinedMesh q) omega j rows4 licensed4
        (savedSelectedRows e chi omega q) t := by
  unfold actualNext5
  apply le_antisymm
  · apply finiteMax_le _ _ _ _ _ (finiteMax_seed_le _ _ _ _)
    intro r hr he
    have howner : r.cell ∈ savedOwners e chi omega q := he.1.symm ▸ hj
    exact finiteMax_row_le _ _ _ _ r (Finset.mem_filter.mpr ⟨hr, howner⟩) he
  · apply finiteMax_le _ _ _ _ _ (finiteMax_seed_le _ _ _ _)
    intro r hr he
    exact finiteMax_row_le _ _ _ _ r (Finset.mem_filter.mp hr).1 he

    (chi omega : ℝ) (homega : 0 < omega) (homegaRho : omega < (19 / 96 : ℝ))

    (hkeys : SavedKeysNodup) (hpositive : SavedPositiveKeys)
    (rows4 : Finset ProducerRow) (licensed4 : ProducerRow → Prop)

    orderedLiteralSum e chi omega q ≤ savedFullIntegral e chi omega q rows4 licensed4 := by
  have hk := saved_key_check_of_nodup hkeys
  have hbound := sum_literalRawOwnerIntegrals_le_fullSelectedIntegral savedRecords
    q hq.positive hq.grid400 hq.nonnegativeMask (loaded_key_injective_finset archive hk)
    (fun raw hr i => hpositive raw (List.mem_toFinset.mp hr) i)
    e he chi omega homega homegaRho rows4 licensed4 hgeometry (savedJ5 q) hq.allGates
  have hsum := sum_loaded_records_eq_finset archive hk
    (fun raw => literalRawOwnerSum raw e chi (refinedMesh q) omega (savedJ5 q))
  change (∑ raw ∈ archive.toList.toFinset,
    literalRawOwnerSum raw e chi (refinedMesh q) omega (savedJ5 q)) ≤
      savedFullIntegral e chi omega q rows4 licensed4 at hbound
  rw [← hsum] at hbound
  exact hbound
