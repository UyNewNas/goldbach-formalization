import Std.Data.TreeMap.Raw.Lemmas
import Std.Data.TreeMap.Raw.WF
import Lean.Elab.Tactic.Omega

namespace P24KernelTablePrefix

universe u

abbrev NodeMap (α : Type u) := Std.TreeMap.Raw Nat α

def sourceState {α : Type u} (initial : α)
    (update : Nat → (Nat → α) → α) : Nat → (Nat → α) :=
  Nat.rec (motive := fun _ => Nat → α) (fun _ => initial)
    (fun n old i => if i = n + 1 then update (n + 1) old else old i)

def mapRead {α : Type u} (initial : α) (nodes : NodeMap α) (i : Nat) : α :=
  nodes.getD i initial

def mapStep {α : Type u} (initial : α)
    (update : Nat → (Nat → α) → α) (next : Nat) (nodes : NodeMap α) : NodeMap α :=
  nodes.insert next (update next (mapRead initial nodes))

def mapBuildFrom {α : Type u} (initial : α)
    (update : Nat → (Nat → α) → α) : Nat → Nat → NodeMap α → NodeMap α
  | 0, _, nodes => nodes
  | remaining + 1, next, nodes =>
      mapBuildFrom initial update remaining (next + 1)
        (mapStep initial update next nodes)

def Represents {α : Type u} (initial : α)
    (update : Nat → (Nat → α) → α) (completed : Nat) (nodes : NodeMap α) : Prop :=
  nodes.WF ∧ mapRead initial nodes = sourceState initial update completed

theorem empty_represents {α : Type u} (initial : α)
    (update : Nat → (Nat → α) → α) :
    Represents initial update 0 (∅ : NodeMap α) := by
  refine ⟨Std.TreeMap.Raw.WF.emptyc, ?_⟩
  funext i
  change (∅ : NodeMap α).getD i initial = initial
  exact Std.TreeMap.Raw.getD_emptyc

theorem mapRead_step {α : Type u} (initial : α)
    (update : Nat → (Nat → α) → α) (next : Nat)
    (nodes : NodeMap α) (hw : nodes.WF) :
    mapRead initial (mapStep initial update next nodes) =
      fun i => if i = next then update next (mapRead initial nodes)
        else mapRead initial nodes i := by
  funext i
  have hcmp : compare next i = Ordering.eq ↔ i = next :=
    Nat.compare_eq_eq.trans eq_comm
  simpa only [mapRead, mapStep, hcmp] using
    (Std.TreeMap.Raw.getD_insert (t := nodes) hw (k := next) (a := i)
      (v := update next (mapRead initial nodes)) (fallback := initial))

theorem mapStep_represents {α : Type u} (initial : α)
    (update : Nat → (Nat → α) → α) (n : Nat)
    (nodes : NodeMap α) (h : Represents initial update n nodes) :
    Represents initial update (n + 1) (mapStep initial update (n + 1) nodes) := by
  constructor
  · exact h.1.insert
  · rw [mapRead_step initial update (n + 1) nodes h.1, h.2]
    rfl

theorem mapBuildFrom_represents {α : Type u} (initial : α)
    (update : Nat → (Nat → α) → α) (remaining n : Nat)
    (nodes : NodeMap α) (h : Represents initial update n nodes) :
    Represents initial update (n + remaining)
      (mapBuildFrom initial update remaining (n + 1) nodes) := by
  induction remaining generalizing n nodes with
  | zero => simpa only [mapBuildFrom, Nat.add_zero] using h
  | succ remaining ih =>
      rw [show n + (remaining + 1) = (n + 1) + remaining by omega]
      exact ih (n + 1) (mapStep initial update (n + 1) nodes)
        (mapStep_represents initial update n nodes h)

theorem mapBuildFrom_empty_lookup {α : Type u} (initial : α)
    (update : Nat → (Nat → α) → α) (count : Nat) :
    mapRead initial (mapBuildFrom initial update count 1 ∅) =
      sourceState initial update count := by
  have h := mapBuildFrom_represents initial update count 0 ∅
    (empty_represents initial update)
  simpa only [Nat.zero_add] using h.2

/-- Later writes cannot change any index already in the completed prefix.
No causality condition on the update function is required. -/
theorem sourceState_completed_index_stable {α : Type u} (initial : α)
    (update : Nat → (Nat → α) → α) (completed remaining i : Nat)
    (hi : i ≤ completed) :
    sourceState initial update (completed + remaining) i =
      sourceState initial update completed i := by
  induction remaining with
  | zero => rfl
  | succ remaining ih =>
      have hne : i ≠ completed + remaining + 1 := by omega
      change (if i = completed + remaining + 1 then
        update (completed + remaining + 1)
          (sourceState initial update (completed + remaining))
        else sourceState initial update (completed + remaining) i) =
          sourceState initial update completed i
      rw [if_neg hne]
      exact ih

/-- Only the first b executed updates, at indices 1 through b, are assumed
constant. In particular this hypothesis imposes no condition on update 0. -/
theorem sourceState_constant_prefix {α : Type u} (initial : α)
    (update : Nat → (Nat → α) → α) (b : Nat)
    (hprefix : ∀ k, k < b → ∀ old, update (k + 1) old = initial) :
    sourceState initial update b = fun _ => initial := by
  revert hprefix
  induction b with
  | zero =>
      intro _
      rfl
  | succ b ih =>
      intro hprefix
      have hprevious : sourceState initial update b = fun _ => initial :=
        ih (fun k hk old => hprefix k (by omega) old)
      funext i
      change (if i = b + 1 then
        update (b + 1) (sourceState initial update b)
        else sourceState initial update b i) = initial
      by_cases hi : i = b + 1
      · rw [if_pos hi]
        exact hprefix b (Nat.lt_succ_self b) (sourceState initial update b)
      · rw [if_neg hi]
        exact congrFun hprevious i

/-- The unchanged default reader represents an entire proved constant prefix;
no stored rows or observed values are supplied. -/
theorem empty_represents_constant_prefix {α : Type u} (initial : α)
    (update : Nat → (Nat → α) → α) (b : Nat)
    (hprefix : ∀ k, k < b → ∀ old, update (k + 1) old = initial) :
    Represents initial update b (∅ : NodeMap α) :=
  ⟨Std.TreeMap.Raw.WF.emptyc,
    (empty_represents initial update).2.trans
      (sourceState_constant_prefix initial update b hprefix).symm⟩

/-- Resume at b+1 using the same update function. The conclusion is equality
of total readers, including zero, unwritten slots and out-of-prefix indices. -/
theorem mapBuildFrom_constant_prefix_lookup {α : Type u} (initial : α)
    (update : Nat → (Nat → α) → α) (b remaining : Nat)
    (hprefix : ∀ k, k < b → ∀ old, update (k + 1) old = initial) :
    mapRead initial (mapBuildFrom initial update remaining (b + 1) ∅) =
      sourceState initial update (b + remaining) :=
  (mapBuildFrom_represents initial update remaining b ∅
    (empty_represents_constant_prefix initial update b hprefix)).2

/-- A completed-index query also agrees with any longer original computation.
This does not assert equality of the two total readers beyond that prefix. -/
theorem mapBuildFrom_constant_prefix_query_stable {α : Type u} (initial : α)
    (update : Nat → (Nat → α) → α) (b remaining later i : Nat)
    (hprefix : ∀ k, k < b → ∀ old, update (k + 1) old = initial)
    (hi : i ≤ b + remaining) :
    mapRead initial (mapBuildFrom initial update remaining (b + 1) ∅) i =
      sourceState initial update (b + remaining + later) i :=
  (congrFun (mapBuildFrom_constant_prefix_lookup initial update b remaining hprefix) i).trans
    (sourceState_completed_index_stable initial update (b + remaining) later i hi).symm
